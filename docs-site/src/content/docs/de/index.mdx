import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components'

<CardGrid>
  <Card title="Sofort gefunden" icon="magnifier">
    Das ganze Verzeichnis liegt im Browser: Namen, Firmen, Telefonnummern in jeder Schreibweise und alle Felder durchsuchen. Verschachtelte Filter, gespeicherte Listen, Favoriten, und als installierte App startet es auch offline.
  </Card>
  <Card title="Immer aktuell" icon="approve-check">
    Jeder Kontakt hat einen Zuständigen, der ihn regelmäßig prüft. Nur-Lesen-Benutzer schlagen Änderungen vor, Bearbeiter entscheiden. Benachrichtigungen aufs Handy, Datenqualität im Verlauf und Aufbewahrungsregeln.
  </Card>
  <Card title="Auf jedem Telefon" icon="phone">
    Kontakte in der Outlook-App, als CardDAV-Adressbücher und als Telefonbuch der Tischtelefone. Telefonanlagen fragen Kartei, wer anruft.
  </Card>
  <Card title="Im Takt mit euren Systemen" icon="random">
    Active Directory, Exchange Online, Single Sign-on (OpenID Connect, SCIM), LDAP, CSV/XLSX-Exporte, Webhooks und eine REST-API.
  </Card>
  <Card title="Eigene Felder" icon="list-format">
    Die Administration legt Felder, Auswahlwerte, Prüfregeln und Sichtbarkeit fest, alles in der Weboberfläche, auf Deutsch und Englisch.
  </Card>
  <Card title="Selbst betrieben und offen" icon="server">
    Docker und PostgreSQL, tägliche Sicherungen mit Wiederherstellungstest, Endpunkte fürs Monitoring und ein Audit-Log. Freie Software unter der AGPL.
  </Card>
</CardGrid>
