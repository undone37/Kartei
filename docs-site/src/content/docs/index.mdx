import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components'

<CardGrid>
  <Card title="Found in an instant" icon="magnifier">
    The whole directory is in the browser: search names, companies, phone numbers in any format and every field. Nested filters, saved lists, favorites, and it starts offline as an installed app.
  </Card>
  <Card title="Kept up to date" icon="approve-check">
    Every contact has a responsible colleague who reviews it regularly. Read-only users propose changes, maintainers decide. Notifications on the phone, a data-quality trend and retention rules.
  </Card>
  <Card title="On every phone" icon="phone">
    Contacts in the Outlook app, as CardDAV address books and as the phonebook of desk phones. Phone systems ask Kartei who is calling.
  </Card>
  <Card title="In sync with your systems" icon="random">
    Active Directory, Exchange Online, single sign-on (OpenID Connect, SCIM), LDAP, CSV/XLSX exports, webhooks and a REST API.
  </Card>
  <Card title="Your own fields" icon="list-format">
    Admins define fields, options, validation and who sees what, all in the web interface, in German and English.
  </Card>
  <Card title="Self-hosted and open" icon="server">
    Docker and PostgreSQL, daily backups with a restore test, monitoring endpoints and an audit log. Free software under the AGPL.
  </Card>
</CardGrid>
